// My Tasks lists, projected so the gallery template reads flat fields rather
// than three lookups per row per property.
//
// THE PROJECTION IS DUPLICATED ON PURPOSE. It was a UDF, nfTskProject(rows:
// Table): Table, and Studio rejected it: "Unknown type Table" and
// "'nfTskProject' is an unknown or unsupported function". UDF parameter and
// return types are scalars only. Named formulas take no parameters, so there is
// no shared-projection route at all, and writing the fifteen lines twice is the
// correct answer rather than a workaround. See house-rules section 7.
//
// DELEGATION. These filter colAssignments, an in-memory collection, so nothing
// here is subject to a delegation limit. The limit is at the seed, in
// App.OnStart, where colAssignments is filled from the list on the indexed
// AssigneeEmail and StatusKey columns. A truncated seed gives a silently short
// task list and wrong counts everywhere downstream. See CLAUDE.md Delegation.
//
// The joins to colRequests and colDepartments are LookUps against collections
// and are local by the same reasoning.
nfTskPending =
    SortByColumns(
        ForAll(
            Filter(
                nfMyAllLive,
                // NOT "<> Completed": <> DOES NOT DELEGATE ON SHAREPOINT, and the
                // warning it raises is real rather than cosmetic, because delegation
                // composes through a named formula: this filter and nfMyAllLive fold
                // into ONE server query, so one non-delegable operator drops the whole
                // thing to the client. An OR of equalities delegates.
                //
                // Safe because StatusKey is a CLOSED SET of four, fixed by the data
                // contract precisely so flow logic can depend on it. If a fifth open
                // status is ever added, this line has to be extended with it.
                StatusKey = "NotStarted" || StatusKey = "InProgress" || StatusKey = "OnHold"
            ) As a,
            //  ONE request fetch per row, from the LIST. It was two LookUps into
            //  colRequests, a global load capped by the data row limit, so a task
            //  whose request fell outside the fetched head rendered a BLANK title
            //  and a blank priority. Raising the limit hid that rather than fixing
            //  it: the cap moved, it did not go away.
            //
            //  RequestKey is indexed, so each fetch delegates. The cost is one
            //  server call per assignment row, which is bounded because this set is
            //  ONE user's work. If it ever needs to be cheaper, denormalise Title
            //  onto RequestAssignments the way PhaseCount is denormalised onto
            //  Requests, and drop the per-row fetch entirely.
            With(
            { r: LookUp(colRequests, RequestKey = a.RequestKey) },
            {
                AssignmentKey:  a.AssignmentKey,
                RequestKey:     a.RequestKey,
                RequestTitle:   r.Title,
                DepartmentCode: a.DepartmentCode,
                DepartmentName: LookUp(nfDepartments, ConfigKey = a.DepartmentCode).Title,
                Goal:           a.Goal,
                AssigneeName:   a.AssigneeName,
                DueDate:        a.DueDate,
                CompletedDate:  a.CompletedDate,
                PriorityKey:    r.PriorityKey,
                IsOverdue:      !IsBlank(a.DueDate) && a.DueDate < Today(),
                DueLabel:
                    If(

// -----------------------------------------------------------------------------
//  DATE BOUNDARIES
//  Contract s6 + CLAUDE.md: SharePoint stores UTC, Canvas displays local. Due
//  Today and Overdue drive the dashboard hero, so day boundaries must be
//  computed in the user's timezone or counts read a day off for some users.
//  Today() is already local, so these are correct for the collections version.
//  SHAREPOINT VERSION: values arriving from SharePoint are UTC. Convert on load

nfToday        = Today();
nfNow          = Now();


// -----------------------------------------------------------------------------
//  SHARED DERIVED READS
//  CLAUDE.md: every read filters IsDeleted = false. Use nfActiveRequests as the
//  base for any request query rather than repeating the filter.
//
//  DELEGATION: these read the LISTS. IsDeleted is indexed and the comparison is a
//  plain equality, so the base filter delegates, and delegation COMPOSES THROUGH
//  A NAMED FORMULA: a derived filter folds into the same server query rather than
//  running over a fetched page. That is the whole reason these are worth having as
//  named formulas instead of collections.
//
//  It also means one non-delegable operator anywhere in a derived chain drops the
//  ENTIRE chain to the client, silently and with no error. Check each addition.
// -----------------------------------------------------------------------------

nfMyTemplateCategories =
    Filter(
        nfCategories,
        !IsBlank(DefaultTemplateKey),
        nfUserEmail in Lower(Coalesce(Admins, "|none|"))
    );
// THE REQUEST TYPES THIS USER MAY EDIT THE BLUEPRINT FOR. Settings shows a
// workflow editor per row of this and hides the whole section when it is empty.
//
// SUBSTRING TEST ON A SEMICOLON LIST, the same idiom canManageRequest uses on
// RequestAdmins, and sufficient for the same reason: this is UX GATING, NOT
// SECURITY. It hides the editor. Anyone with list access can edit
// WorkflowTemplates directly in SharePoint. Do not describe it as a control.
//
// COALESCE TO A SENTINEL, NOT TO "". A blank Admins with Coalesce(Admins, "")
// becomes `nfUserEmail in ""`, and `in` against an empty string is TRUE for
// every user, which would open the editor on exactly the categories nobody
// administers. That is the precise inversion of what a blank permission column
// must mean. "|none|" cannot occur in an email address, so it matches nobody.
//
// DefaultTemplateKey MUST BE PRESENT TOO. A category with an admin but no
// template has no blueprint to edit.
//
// ORDER OF OPERATIONS, LEARNED THE EXPENSIVE WAY. Referencing Admins in the
// ShowColumns above BEFORE the column existed made colConfig an ERROR rather
// than a table, and colCategories, colDepartments and colProcessingTypes all
// derive from it, so every label lookup in the app failed at once. Creating the
// column in SharePoint is not enough either: the app caches the list schema
// inside the msapp, so the data source must be REFRESHED and republished in
// Studio before the name resolves. Column, then refresh, then this reference.

nfActiveRequests = Filter(colRequests, IsDeleted = false);

// ---------------------------------------------------------------------------
//  LIVE PER-USER SETS. These read the LIST, not a collection, so My Tasks is
//  always current and does not depend on the App.OnStart snapshot.
//
//  AssigneeEmail and StatusKey are both indexed, so the fetch delegates.
//  NOTE THE ABSENCE OF Lower(): a function on the COLUMN is not delegable and
//  would silently drop the query to the client. Emails are written lower-cased
//  at the source, which is what makes the plain comparison safe.
//
//  The date buckets below are counted IN MEMORY off nfMyOpenLive rather than
//  filtered server side. That is deliberate: one user's open work is tens of
//  rows, so the counts are exact, and it avoids depending on date delegation.
//
//  These are per-user counts ONLY. Global aggregates stay in colMetrics because
//  CountRows does not delegate on SharePoint, so a tenant-wide count would
//  silently total just the fetched page.
// ---------------------------------------------------------------------------

//  ONE delegable query serves the whole screen. AssigneeEmail is indexed and the
//  comparison is a plain equality, which is the least ambiguous thing to delegate.
//  CORRECTION, measured in Studio: there is no "in memory" split here. DELEGATION
//  COMPOSES THROUGH A NAMED FORMULA, so this filter and the one below fold into a
//  SINGLE server query. That is why a non-delegable operator in the derived formula
//  raises a real warning rather than a cosmetic one.
//
//  <> IS NOT DELEGABLE ON SHAREPOINT. It was assumed to be, and it is not: changing
//  it to = made the warning disappear, which is what proved both facts at once. The
//  open set is therefore written as an OR of equalities over the closed status set.
nfMyAllLive = Filter(colAssignments, AssigneeEmail = nfUserEmail);

nfMyOpenLive = Filter(nfMyAllLive, StatusKey = "NotStarted" || StatusKey = "InProgress" || StatusKey = "OnHold");
nfMyDoneLive = Filter(nfMyAllLive, StatusKey = "Completed");

nfMyOpenCount     = CountRows(nfMyOpenLive);
nfMyDoneCount     = CountRows(nfMyDoneLive);
nfMyOverdueCount  = CountRows(Filter(nfMyOpenLive, !IsBlank(DueDate), DueDate < Today()));
nfMyDueTodayCount = CountRows(Filter(nfMyOpenLive, !IsBlank(DueDate), DueDate = Today()));
nfMyUpcomingCount = CountRows(Filter(nfMyOpenLive, !IsBlank(DueDate), DueDate > Today()));

//  nfMyAssignments and nfMyOpenAssignments are GONE. They read colAssignments and
//  applied Lower() TO THE COLUMN, which is not delegable, and they duplicated
//  nfMyOpenLive above. The dashboard now shares My Tasks' set, so the two screens
//  cannot disagree about what the user's open work is.

//  Requests this user owns. No Lower() on the column, for the reason given above;
//  SharePoint text comparison is case insensitive, so the plain equality is both
//  correct and delegable. PrimaryPocEmail is NOT an indexed column: that does not
//  affect delegation, it only means SharePoint has no index to serve it from.
nfMyRequests = Filter(colRequests, IsDeleted = false, PrimaryPocEmail = nfUserEmail);

//  CountRows does not delegate, so this totals the fetched page rather than the
//  list. Safe here ONLY because it is scoped to one person's requests, which is
//  the same reasoning as the per-user counts above. It is NOT safe for a
//  tenant-wide total, which is what Metrics is for.
nfMyRequestsCount = CountRows(nfMyRequests);

//  Sentinel floor for "has a due date". !IsBlank() compiles to Not(), which does
//  NOT delegate and would drop a whole chain to the client. A date comparison
//  does delegate, and a blank date cannot be greater than the floor, so this
//  excludes blanks server side and expresses the same intent.
nfDateFloor = Date(2000, 1, 1);

// scrHome, urgent hero. Due today OR already overdue.
// The mockup's hero says "due today" and filters to exactly today, which hides
// overdue work on the landing screen. Overdue is strictly more urgent than due
// today, so it is included here and the heading reads "need your attention"
// rather than "due today". Deliberate deviation from the mockup wording.
//  Fully delegable: both predicates are date comparisons and the base set is one
//  indexed equality. nfDateFloor replaces !IsBlank(DueDate) because Not() does not
//  delegate, and a non-delegable operator here would have folded back into
//  nfMyOpenLive and capped the hero at the data row limit. A user with more open
//  assignments than the limit would then have seen an INCOMPLETE urgent list with
//  no error anywhere, which is the failure mode worth engineering against.
nfDashUrgent = SortByColumns(
    Filter(nfMyOpenLive, DueDate > nfDateFloor, DueDate <= nfToday),
    "DueDate",
    SortOrder.Ascending
);

//  Counted ONCE here rather than three times on the screen. The hero container's
//  Visible, its heading, and the caught-up empty state all need the same number,
//  and each CountRows on a screen control is its own round trip. A named formula
//  also guarantees the three cannot disagree.
//
//  NOT galDashUrgent.AllItemsCount, which would be cheaper but circular: the
//  gallery lives INSIDE the container whose Visible this drives, and a hidden
//  container's gallery has nothing loaded, so the hero could never appear.
//  DEFAULT VIEW, read from the LIST rather than from colUserPrefs.
//
//  StartScreen is evaluated BEFORE App.OnStart runs, so it cannot see any
//  collection. A named formula can be read there, and this one reads the list
//  directly, so the preference is honoured on the very first paint rather than
//  one navigation later.
//
//  UserEmail is indexed, so the LookUp delegates. It costs one single-row query
//  at startup, which is the price of the setting working at all.
nfDefaultView = Coalesce(
    LookUp(colUserPrefs, UserEmail = Lower(nfUserEmail)).DefaultView,
    "Home"
);

nfDashUrgentCount = CountRows(nfDashUrgent);

// scrHome, recent requests. Newest five.
//
//  FirstN is not delegable, and it does not need to be. The SORT delegates, so the
//  server returns the newest rows first and the first page already contains the
//  five newest in the whole list. Taking five off the front of a server-sorted
//  page is exact, not approximate.
//
//  Sort on "Created", the built-in column. It was "CreatedDate" once, which is not
//  a column on this list, and the mistake silently emptied this gallery rather
//  than erroring.
nfDashRecent = FirstN(
    SortByColumns(nfActiveRequests, "Created", SortOrder.Descending),
    5
);

// -----------------------------------------------------------------------------
//  ROUTING
//  One routing table for the whole app. Both navs call this rather than each
//  carrying its own Switch, so a new screen is one line in one place.
//
//  This is a BEHAVIOUR user defined function: it takes a parameter and wraps
//  its body in braces, per house-rules s7. Named formulas cannot contain
//  Navigate; behaviour UDFs can. First use of the construct in this project,
//  so confirm it binds in Studio before relying on it elsewhere.

nfRoute(navID: Number): Void = {
    Switch(
        navID,

        7, Navigate(scrAllRequests, ScreenTransition.None, { locMyRequestsOnly: true }),
        4, Navigate(scrReports,     ScreenTransition.None),
        5, Navigate(scrSettings,    ScreenTransition.None),
        6, Navigate(scrHelp,        ScreenTransition.None)
    )
};
